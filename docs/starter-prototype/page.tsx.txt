"use client";

import { useMemo, useState } from "react";
import items from "@/data/zoho-items.json";
import {
  Activity, ArrowDownToLine, ArrowUpRight, CalendarDays, Check, CheckCircle2,
  ChevronLeft, ChevronRight, CircleHelp, Clock3, Factory, FileSpreadsheet,
  LayoutDashboard, LogOut, Package, Search, ShieldCheck, Users, X
} from "lucide-react";

type Item = { zoho_item_id: string; name: string; part_code: string | null; unit: string | null; unit_needs_review: boolean };
const catalogue = items as Item[];

export default function Home() {
  const [role, setRole] = useState<"operator" | "customer" | "manager">("operator");
  const [entryMeta, setEntryMeta] = useState<Record<string, { at: string; by: string }>>({});
  const [query, setQuery] = useState("");
  const [page, setPage] = useState(1);
  const [quantities, setQuantities] = useState<Record<string,string>>({});
  const [notice, setNotice] = useState("");
  const [published, setPublished] = useState(false);
  const pageSize = 12;
  const filtered = useMemo(() => catalogue.filter((item) =>
    `${item.name} ${item.part_code ?? ""}`.toLowerCase().includes(query.toLowerCase())
  ), [query]);
  const pageCount = Math.max(1, Math.ceil(filtered.length / pageSize));
  const visible = filtered.slice((page - 1) * pageSize, page * pageSize);
  const completed = catalogue.filter((item) => quantities[item.zoho_item_id] !== undefined && quantities[item.zoho_item_id] !== "").length;
  const ready = completed === catalogue.length && catalogue.every((i) => !i.unit_needs_review);
  const update = (id: string, value: string) => {
    if (value !== "" && (!/^\d+(\.\d{1,3})?$/.test(value) || Number(value) < 0)) return;
    setQuantities((old) => ({ ...old, [id]: value }));
    if (value !== "") setEntryMeta((old) => ({ ...old, [id]: { at: new Date().toISOString(), by: "Stock operator (demo)" } }));
    else setEntryMeta((old) => { const next = { ...old }; delete next[id]; return next; });
    setPublished(false);
  };
  const exportCsv = () => {
    const rows = [["Part name","Part code","Unit","Available quantity","Zoho item ID"],
      ...catalogue.map(i => [i.name, i.part_code ?? "", i.unit ?? "UNIT REQUIRED", quantities[i.zoho_item_id] ?? "", i.zoho_item_id])];
    const csv = rows.map(row => row.map(v => `"${String(v).replaceAll('"','""')}"`).join(",")).join("\n");
    const url = URL.createObjectURL(new Blob([csv], { type: "text/csv;charset=utf-8" }));
    const a = document.createElement("a"); a.href = url; a.download = "apt-daily-stock-draft.csv"; a.click(); URL.revokeObjectURL(url);
  };
  const publish = () => {
    if (completed !== catalogue.length) { setNotice(`Add quantities for all ${catalogue.length} goods items before publishing.`); return; }
    if (catalogue.some(i => i.unit_needs_review)) { setNotice("An administrator must assign units to the 60 items with missing units before publishing."); return; }
    setPublished(true); setNotice("Demo report published in this browser session. Connect Supabase to publish securely for customers.");
  };

  return <main className="shell">
    <aside className="sidebar">
      <div className="brand">
        <img src="/apt-mark.png" alt="APT logo" className="brand-mark"/>
        <div><div className="brand-name">AXIS PLASTICS</div><div className="brand-sub">TRANSITION · APT</div></div>
      </div>
      <div className="workspace-label">WORKSPACE</div>
      <button className="nav-item active"><LayoutDashboard size={17}/> Stock overview</button>
      <button className="nav-item" onClick={() => setRole("operator")}><Package size={17}/> Daily stock entry</button>
      <button className="nav-item" onClick={exportCsv}><FileSpreadsheet size={17}/> Export catalogue</button>
      <div className="sidebar-bottom">
        <div className="secure-note"><ShieldCheck size={17}/><span>Private factory workspace<br/><small>Role-based access planned</small></span></div>
        <div className="user-chip"><div className="avatar">APT</div><div><b>{role === "operator" ? "Stock operator" : role === "manager" ? "Factory manager" : "Customer view"}</b><small>{role === "operator" ? "Employee portal" : role === "manager" ? "Management portal" : "Read-only portal"}</small></div><LogOut size={15} className="muted"/></div>
      </div>
    </aside>

    <section className="main">
      <header className="topbar">
        <div className="mobile-brand"><img src="/apt-mark.png" alt="APT"/><b>APT STOCK PORTAL</b></div>
        <div className="crumb">Operations <span>/</span> {role === "operator" ? "Daily stock entry" : role === "manager" ? "Manager activity view" : "Customer dashboard"}</div>
        <div className="top-actions"><span className="demo-pill"><span/> DEMO PREVIEW</span><button className="icon-button" aria-label="Help"><CircleHelp size={18}/></button></div>
      </header>

      <div className="content">
        <div className="welcome-row">
          <div>
            <div className="eyebrow"><Factory size={14}/> AXIS PLASTICS TRANSITION</div>
            <h1>{role === "operator" ? "Daily stock entry" : role === "manager" ? "Entry activity & timestamps" : "Stock availability"}</h1>
            <p>{role === "operator" ? "Record today's available finished-goods stock across the catalogue." : role === "manager" ? "Review who entered each quantity and exactly when it was recorded." : "View the latest published finished-goods stock report."}</p>
          </div>
          <div className="date-card"><CalendarDays size={17}/><div><small>REPORTING DATE</small><b>{new Intl.DateTimeFormat("en-IN",{day:"2-digit",month:"short",year:"numeric"}).format(new Date())}</b></div></div>
        </div>

        <div className="stats-grid">
          <div className="stat-card"><div className="stat-icon blue"><Package size={18}/></div><div><small>GOODS ITEMS</small><strong>235</strong><span>Imported from Zoho Books</span></div></div>
          <div className="stat-card"><div className="stat-icon green"><CheckCircle2 size={18}/></div><div><small>QUANTITIES ENTERED</small><strong>{completed}<em> / 235</em></strong><span>{Math.round(completed / 235 * 100)}% complete</span></div></div>
          <div className="stat-card"><div className="stat-icon amber"><CircleHelp size={18}/></div><div><small>UNIT REVIEW REQUIRED</small><strong>60</strong><span>Missing in Zoho export</span></div></div>
          <div className="stat-card"><div className="stat-icon violet"><Clock3 size={18}/></div><div><small>REPORT STATUS</small><strong className="status-text">{published ? "Published*" : "Draft"}</strong><span>{published ? "Local demo only" : "Not yet submitted"}</span></div></div>
        </div>

        <div className="progress-card">
          <div className="progress-head"><div><b>Daily report completion</b><span>Every goods item needs a quantity, including zero-stock items.</span></div><strong>{completed}<small> / 235</small></strong></div>
          <div className="progress-track"><div style={{width:`${completed/235*100}%`}}/></div>
        </div>

        <div className="table-card">
          <div className="table-head">
            <div><h2>{role === "operator" ? "Finished-goods catalogue" : role === "manager" ? "Stock entry register" : "Available stock"}</h2><p>{role === "manager" ? "Review the operator and timestamp recorded against each stock quantity" : "Part details synced from the Zoho Books item export"}</p></div>
            <div className="table-actions"><div className="search"><Search size={16}/><input value={query} onChange={e=>{setQuery(e.target.value);setPage(1)}} placeholder="Search name or part code"/></div><button className="secondary-button" onClick={exportCsv}><ArrowDownToLine size={16}/> Export CSV</button></div>
          </div>
          <div className="table-wrap"><table><thead><tr><th>PART / ITEM</th><th>PART CODE</th><th>UNIT</th><th>{role === "operator" ? "AVAILABLE QUANTITY" : "AVAILABLE STOCK"}</th>{role === "manager" && <th>ENTERED BY / TIME</th>}<th>DATA CHECK</th></tr></thead>
            <tbody>{visible.map(item=><tr key={item.zoho_item_id}>
              <td><div className="part-name">{item.name}</div><div className="part-id">Zoho ID · {item.zoho_item_id}</div></td>
              <td><span className="code-tag">{item.part_code || "—"}</span></td>
              <td>{item.unit ? <span className="unit-tag">{item.unit}</span> : <span className="warning-tag">Set unit</span>}</td>
              <td>{role === "operator" ? <input className="qty-input" inputMode="decimal" value={quantities[item.zoho_item_id] ?? ""} onChange={e=>update(item.zoho_item_id,e.target.value)} placeholder="Enter qty" disabled={item.unit_needs_review} aria-label={`Available quantity for ${item.name}`}/> : <b className="stock-value">{(published || role === "manager") ? quantities[item.zoho_item_id] ?? "—" : "—"}</b>}</td>
              {role === "manager" && <td>{entryMeta[item.zoho_item_id] ? <><div className="part-name">{entryMeta[item.zoho_item_id].by}</div><div className="part-id">{new Intl.DateTimeFormat("en-IN", { dateStyle: "medium", timeStyle: "short" }).format(new Date(entryMeta[item.zoho_item_id].at))}</div></> : <span className="pending-tag">No entry yet</span>}</td>}
              <td>{item.unit_needs_review ? <span className="warning-tag">Unit review</span> : quantities[item.zoho_item_id] !== undefined && quantities[item.zoho_item_id] !== "" ? <span className="done-tag"><Check size={12}/> Entered</span> : <span className="pending-tag">Pending</span>}</td>
            </tr>)}</tbody></table>
          </div>
          <div className="table-foot"><span>Showing {filtered.length ? (page-1)*pageSize+1 : 0}–{Math.min(page*pageSize,filtered.length)} of {filtered.length} items</span><div className="pagination"><button disabled={page===1} onClick={()=>setPage(p=>p-1)}><ChevronLeft size={16}/></button><span>Page {page} of {pageCount}</span><button disabled={page===pageCount} onClick={()=>setPage(p=>p+1)}><ChevronRight size={16}/></button></div></div>
        </div>

        {role === "operator" && <div className="bottom-actions"><div className="note"><ShieldCheck size={17}/><span><b>Review before publishing</b><small>Unit setup is required for 60 items before quantities can be submitted.</small></span></div><div className="action-buttons"><button className="secondary-button" onClick={()=>setNotice("Draft is currently held in this browser session only. Supabase persistence will be enabled in the connected build.")}>Save draft</button><button className="primary-button" onClick={publish}>Review & publish <ArrowUpRight size={16}/></button></div></div>}
        {notice && <div className="notice"><span>{notice}</span><button onClick={()=>setNotice("")} aria-label="Dismiss"><X size={16}/></button></div>}

        <div className="demo-switch"><span>Preview another role</span><button className={role==="operator"?"selected":""} onClick={()=>setRole("operator")}>Operator</button><button className={role==="manager"?"selected":""} onClick={()=>setRole("manager")}>Manager</button><button className={role==="customer"?"selected":""} onClick={()=>setRole("customer")}>Customer</button><span className="demo-warning">Demo only · timestamps are local preview data</span></div>
        <footer>© {new Date().getFullYear()} AXIS PLASTICS TRANSITION <span>FACTORY STOCK PORTAL · V1 CONCEPT</span></footer>
      </div>
    </section>
  </main>;
}
